// Helper function to set color
void setColor(Color color) {
    glColor4f(color.r, color.g, color.b, color.a);
}

// Enhanced rectangle with gradients
void drawRect(float x, float y, float width, float height, Color color, bool gradient = false) {
    if (gradient) {
        glBegin(GL_QUADS);
        setColor(Color(color.r * 1.1f, color.g * 1.1f, color.b * 1.1f, color.a));
        glVertex2f(x, y + height);
        glVertex2f(x + width, y + height);
        setColor(Color(color.r * 0.9f, color.g * 0.9f, color.b * 0.9f, color.a));
        glVertex2f(x + width, y);
        glVertex2f(x, y);
        glEnd();
    } else {
        setColor(color);
        glBegin(GL_QUADS);
        glVertex2f(x, y);
        glVertex2f(x + width, y);
        glVertex2f(x + width, y + height);
        glVertex2f(x, y + height);
        glEnd();
    }
}

// Enhanced circle drawing
void drawCircle(float centerX, float centerY, float radius, Color color, bool glow = false) {
    if (glow) {
        setColor(Color(color.r, color.g, color.b, 0.1f));
        glBegin(GL_TRIANGLE_FAN);
        glVertex2f(centerX, centerY);
        for (int i = 0; i <= 64; i++) {
            float angle = 2.0f * M_PI * i / 64;
            glVertex2f(centerX + cos(angle) * radius * 1.3f, centerY + sin(angle) * radius * 1.3f);
        }
        glEnd();
    }

    setColor(color);
    glBegin(GL_TRIANGLE_FAN);
    glVertex2f(centerX, centerY);
    for (int i = 0; i <= 64; i++) {
        float angle = 2.0f * M_PI * i / 64;
        glVertex2f(centerX + cos(angle) * radius, centerY + sin(angle) * radius);
    }
    glEnd();
}

// Enhanced sky
void drawEnhancedSky() {
    // Gradient sky
    glBegin(GL_QUADS);
    Color topSky(0.6f, 0.8f, 0.95f);
    Color bottomSky(0.9f, 0.95f, 0.98f);

    setColor(topSky);
    glVertex2f(0, windowHeight);
    glVertex2f(windowWidth, windowHeight);
    setColor(bottomSky);
    glVertex2f(windowWidth, windowHeight * 0.4f);
    glVertex2f(0, windowHeight * 0.4f);
    glEnd();

    // Moving clouds
    cloudTime += 0.3f;
    for (int i = 0; i < 5; i++) {
        float cloudX = -150 + (i * 280) + sin(cloudTime * 0.01f + i) * 25;
        float cloudY = windowHeight * 0.7f + cos(cloudTime * 0.008f + i * 2) * 30;

        if (cloudX > windowWidth + 100) cloudX -= windowWidth + 400;

        drawCircle(cloudX, cloudY, 40, GHIBLI_WHITE, true);
        drawCircle(cloudX + 25, cloudY + 8, 32, GHIBLI_WHITE, true);
        drawCircle(cloudX - 20, cloudY - 4, 30, GHIBLI_WHITE, true);
    }
}

// Enhanced tree with better animation

    windTime += 0.2f;

}

// Enhanced student with better walking animation
void drawEnhancedStudent(float x, float y, bool walking = false, float scale = 1.0f, int direction = 1) {
    glPushMatrix();
    glTranslatef(x, y, 0);

    if (direction == -1) {
        glScalef(-1, 1, 1);
    }

    float walkCycle = 0.0f;
    float bobHeight = 0.0f;

    if (walking) {
        walkCycle = sin(animationProgress * 0.012f) * 0.4f;
        bobHeight = abs(sin(animationProgress * 0.024f)) * 3.0f * scale;
        glTranslatef(0, bobHeight, 0);
    }

    float headRadius = 20 * scale;
    float bodyWidth = 26 * scale;
    float bodyHeight = 42 * scale;

    // Shadow
    setColor(GHIBLI_SHADOW);
    drawCircle(0, -5, 32 * scale, GHIBLI_SHADOW);

    // Head
    drawCircle(0, bodyHeight + headRadius, headRadius, GHIBLI_PEACH, true);

    // Hair
    Color hairColor(0.3f, 0.15f, 0.05f);
    drawCircle(-2 * scale, bodyHeight + headRadius + 6 * scale, headRadius * 0.85f, hairColor, true);
    drawCircle(6 * scale, bodyHeight + headRadius + 4 * scale, headRadius * 0.5f, hairColor);

    // Eyes
    drawCircle(-4 * scale, bodyHeight + headRadius + 2 * scale, 2.5 * scale, Color(0.2f, 0.1f, 0.05f));
    drawCircle(4 * scale, bodyHeight + headRadius + 2 * scale, 2.5 * scale, Color(0.2f, 0.1f, 0.05f));

    // Body
    drawRect(-bodyWidth/2, 0, bodyWidth, bodyHeight, GHIBLI_SOFT_BLUE, true);

    // Arms with walking animation
    float armSwing = walking ? walkCycle * 30 * scale : 0;

    glPushMatrix();
    glTranslatef(-bodyWidth/2 - 6 * scale, bodyHeight - 18 * scale, 0);
    glRotatef(armSwing, 0, 0, 1);
    drawRect(0, 0, 10 * scale, 32 * scale, GHIBLI_PEACH);
    glPopMatrix();

    glPushMatrix();
    glTranslatef(bodyWidth/2 - 4 * scale, bodyHeight - 18 * scale, 0);
    glRotatef(-armSwing, 0, 0, 1);
    drawRect(0, 0, 10 * scale, 32 * scale, GHIBLI_PEACH);
    glPopMatrix();

    // Legs with walking animation
    float legSwing = walking ? -walkCycle * 25 * scale : 0;

    glPushMatrix();
    glTranslatef(-6 * scale, -28 * scale, 0);
    glRotatef(legSwing, 0, 0, 1);
    drawRect(0, 0, 10 * scale, 28 * scale, Color(0.2f, 0.2f, 0.4f));
    glPopMatrix();

    glPushMatrix();
    glTranslatef(4 * scale, -28 * scale, 0);
    glRotatef(-legSwing, 0, 0, 1);
    drawRect(0, 0, 10 * scale, 28 * scale, Color(0.2f, 0.2f, 0.4f));
    glPopMatrix();

    // Backpack
    drawRect(-16 * scale, bodyHeight - 22 * scale, 22 * scale, 18 * scale, GHIBLI_SOFT_RED, true);

    glPopMatrix();
}

// Enhanced building with ground connection
void drawEnhancedBuilding(float x, float y, float width, float height, Color color, Color windowColor, int floors = 3) {
    // Building on ground level
    drawRect(x, y, width, height, color, true);

    // Building outline
    setColor(Color(color.r * 0.8f, color.g * 0.8f, color.b * 0.8f));
    glLineWidth(1.5f);
    glBegin(GL_LINE_LOOP);
    glVertex2f(x, y);
    glVertex2f(x + width, y);
    glVertex2f(x + width, y + height);
    glVertex2f(x, y + height);
    glEnd();

    // Windows
    float windowWidth = width / 8.0f;
    float windowHeight = height / (floors * 2.5f);

    for (int floor = 0; floor < floors; floor++) {
        for (int window = 0; window < 4; window++) {
            float wx = x + windowWidth + window * windowWidth * 1.6f;
            float wy = y + height * 0.15f + floor * (height / floors);

            drawRect(wx, wy, windowWidth, windowHeight, windowColor, true);

            // Random lights in windows
            if ((floor + window + (int)(animationProgress * 0.001f)) % 4 == 0) {
                setColor(Color(1.0f, 0.9f, 0.6f, 0.8f));
                drawRect(wx + 2, wy + 2, windowWidth - 4, windowHeight - 4, Color(1.0f, 0.9f, 0.6f, 0.8f));
            }
        }
    }
}

        // Benches every 120 units
        for (int i = 0; i < (int)(length / 140); i++) {

        case 6: drawEnhancedSeminarHall(); break;

    glVertex2f(windowWidth, windowHeight - 80);
    glVertex2f(0, windowHeight - 80);
    glEnd();

    glBegin(GL_QUADS);

    glEnd();

    glutSwapBuffers();
}

// Timer function
void timer(int value) {
    if (autoAnimation) {
        sceneTimer += 16.0f;
        animationProgress += 16.0f;
        windTime += 0.3f;
        cloudTime += 0.2f;

        if (sceneTimer > SCENE_DURATION) {
            currentScene = (currentScene + 1) % totalScenes;
            sceneTimer = 0.0f;
            animationProgress = 0.0f;
        }
    } else {
        animationProgress += 16.0f;
        windTime += 0.3f;
        cloudTime += 0.2f;
    }

    glutPostRedisplay();
    glutTimerFunc(16, timer, 0);
}

// Keyboard input
void keyboard(unsigned char key, int x, int y) {
    switch(key) {
        case 'n':
        case 'N':
            currentScene = (currentScene + 1) % totalScenes;
            animationProgress = 0.0f;
            sceneTimer = 0.0f;
            std::cout << "Moving to: " << sceneNames[currentScene] << std::endl;
            break;
        case 'p':
        case 'P':
            currentScene = (currentScene - 1 + totalScenes) % totalScenes;
            animationProgress = 0.0f;
            sceneTimer = 0.0f;
            std::cout << "Moving to: " << sceneNames[currentScene] << std::endl;
            break;
        case 'a':
        case 'A':
            autoAnimation = !autoAnimation;
            std::cout << "Auto animation: " << (autoAnimation ? "ON" : "OFF") << std::endl;
            break;
        case '1': case '2': case '3': case '4': case '5':
        case '6': case '7': case '8': case '9':
            {
                int newScene = key - '1';
                if (newScene < totalScenes) {
                    currentScene = newScene;
                    animationProgress = 0.0f;
                    sceneTimer = 0.0f;
                    std::cout << "Jumping to: " << sceneNames[currentScene] << std::endl;
                }
            }
            break;
        case 27: // ESC
            std::cout << "Thank you for the Enhanced Journey to CIU!" << std::endl;
            exit(0);
            break;
    }
    glutPostRedisplay();
}

// Initialize OpenGL
void init() {
    glClearColor(0.6f, 0.8f, 0.95f, 1.0f);
    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    gluOrtho2D(0, windowWidth, 0, windowHeight);
    glMatrixMode(GL_MODELVIEW);

    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
    glEnable(GL_LINE_SMOOTH);
    glEnable(GL_POINT_SMOOTH);
    glHint(GL_LINE_SMOOTH_HINT, GL_NICEST);
    glHint(GL_POINT_SMOOTH_HINT, GL_NICEST);
}

// Reshape function
void reshape(int w, int h) {
    windowWidth = w;
    windowHeight = h;
    glViewport(0, 0, w, h);
    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    gluOrtho2D(0, w, 0, h);
    glMatrixMode(GL_MODELVIEW);
}

// Mouse interaction
void mouse(int button, int state, int x, int y) {
    if (state == GLUT_DOWN) {
        if (button == GLUT_LEFT_BUTTON) {
            currentScene = (currentScene + 1) % totalScenes;
            animationProgress = 0.0f;
            sceneTimer = 0.0f;
            std::cout << "Mouse: Moving to " << sceneNames[currentScene] << std::endl;
        } else if (button == GLUT_RIGHT_BUTTON) {
            autoAnimation = !autoAnimation;
            std::cout << "Mouse: Auto animation " << (autoAnimation ? "ON" : "OFF") << std::endl;
        }
        glutPostRedisplay();
    }
}

// Special keys
void specialKeys(int key, int x, int y) {
    switch(key) {
        case GLUT_KEY_RIGHT:
            currentScene = (currentScene + 1) % totalScenes;
            animationProgress = 0.0f;
            sceneTimer = 0.0f;
            break;
        case GLUT_KEY_LEFT:
            currentScene = (currentScene - 1 + totalScenes) % totalScenes;
            animationProgress = 0.0f;
            sceneTimer = 0.0f;
            break;
        case GLUT_KEY_HOME:
            currentScene = 0;
            animationProgress = 0.0f;
            sceneTimer = 0.0f;
            break;
        case GLUT_KEY_END:
            currentScene = totalScenes - 1;
            animationProgress = 0.0f;
            sceneTimer = 0.0f;
            break;
    }
    glutPostRedisplay();
}
